cmake_minimum_required(VERSION 3.24)
project(solar_whitted_cpp VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Windows.h otherwise defines min/max macros that corrupt calls such as
# std::max in the weather, Embree, and OpenGL translation units.
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

find_package(embree 4 CONFIG REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(GLEW REQUIRED)
find_package(imgui CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(cpr CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(highs CONFIG REQUIRED)

add_library(solar_mesh src/stl_mesh.cpp)
target_include_directories(solar_mesh PUBLIC include)

add_library(solar_trace src/trace_scene.cpp src/simulation.cpp src/placement_surface.cpp)
target_include_directories(solar_trace PUBLIC include)
target_link_libraries(solar_trace PUBLIC solar_mesh embree)

add_library(solar_weather src/weather.cpp)
target_include_directories(solar_weather PUBLIC include)
target_link_libraries(solar_weather PUBLIC cpr::cpr nlohmann_json::nlohmann_json)

add_library(solar_race src/race.cpp src/optimizer.cpp)
target_include_directories(solar_race PUBLIC include)
target_link_libraries(solar_race PUBLIC solar_trace solar_weather highs::highs nlohmann_json::nlohmann_json)

add_library(solar_viewer src/viewer.cpp)
target_include_directories(solar_viewer PUBLIC include)
target_link_libraries(solar_viewer PUBLIC solar_race glfw glm::glm GLEW::GLEW imgui::imgui OpenGL::GL)
if(WIN32)
  target_link_libraries(solar_viewer PRIVATE comdlg32)
endif()

add_executable(solar_whitted_cpp src/main.cpp)
target_link_libraries(solar_whitted_cpp PRIVATE solar_viewer)
target_compile_definitions(solar_whitted_cpp PRIVATE SOLAR_DEFAULT_RACE_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/race.json")

if(MSVC)
  target_compile_options(solar_mesh PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(solar_trace PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(solar_weather PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(solar_race PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(solar_viewer PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(solar_whitted_cpp PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(solar_mesh PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(solar_trace PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(solar_weather PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(solar_race PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(solar_viewer PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(solar_whitted_cpp PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(solar_core_tests tests/core_tests.cpp)
  target_link_libraries(solar_core_tests PRIVATE solar_trace solar_weather)
  if(MSVC)
    target_compile_options(solar_core_tests PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(solar_core_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME solar_core_tests COMMAND solar_core_tests)
  add_executable(solar_race_tests tests/race_tests.cpp)
  target_link_libraries(solar_race_tests PRIVATE solar_race)
  add_test(NAME solar_race_tests COMMAND solar_race_tests)
  set_tests_properties(solar_race_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_executable(solar_optimizer_tests tests/optimizer_tests.cpp)
  target_link_libraries(solar_optimizer_tests PRIVATE solar_race)
  add_test(NAME solar_optimizer_tests COMMAND solar_optimizer_tests)
  add_executable(solar_placement_audit_tests tests/placement_audit_tests.cpp)
  target_link_libraries(solar_placement_audit_tests PRIVATE solar_race)
  add_test(NAME solar_placement_audit_tests COMMAND solar_placement_audit_tests)
  add_executable(solar_placement_surface_tests tests/placement_surface_tests.cpp)
  target_link_libraries(solar_placement_surface_tests PRIVATE solar_trace)
  add_test(NAME solar_placement_surface_tests COMMAND solar_placement_surface_tests)
endif()
